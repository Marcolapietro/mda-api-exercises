# Exercise 8: CRUD Endpoints - Notes API (Windows PowerShell 5.1)
# Start the API first: python app.py   (it starts with 3 seed notes: ids 1, 2, 3)
# Paste the blocks in order. Step 0 defines a helper used by the other steps.

# 0. Base URL + helper that prints status code, key headers and body.
#    Windows PowerShell 5.1 THROWS on 4xx/5xx, so the helper uses try/catch to show them too.
$baseUrl = "http://127.0.0.1:5000"

function Show-Http {
    param(
        [string]$Method,
        [string]$Uri,
        [string]$Body,
        [string]$ContentType = "application/json"
    )
    Write-Host ("=" * 70)
    Write-Host "$Method $Uri"
    $params = @{ Method = $Method; Uri = $Uri; UseBasicParsing = $true }
    if ($Body) { $params.Body = $Body; $params.ContentType = $ContentType }
    try {
        $r = Invoke-WebRequest @params
        Write-Host ("Status: {0}" -f [int]$r.StatusCode)
        foreach ($h in "Content-Type", "Location") {
            if ($r.Headers[$h]) { Write-Host ("{0}: {1}" -f $h, $r.Headers[$h]) }
        }
        if ($r.Content) { Write-Host $r.Content } else { Write-Host "(empty body)" }
    } catch {
        $resp = $_.Exception.Response
        if ($resp -eq $null) { throw }
        Write-Host ("Status: {0}" -f [int]$resp.StatusCode)
        Write-Host ("Content-Type: {0}" -f $resp.ContentType)
        if ($resp.Headers["Allow"]) { Write-Host ("Allow: {0}" -f $resp.Headers["Allow"]) }
        Write-Host $_.ErrorDetails.Message
    }
}

# 1. List notes (newest first) -> 200. Invoke-RestMethod turns the JSON into objects.
$list = Invoke-RestMethod -Method Get -Uri "$baseUrl/notes"
$list.data | Format-Table id, title, updated_at | Out-String | Write-Host
Write-Host ("page: " + ($list.page | ConvertTo-Json -Compress))

# 2. List with a page size -> 200 (default 20, values above 100 are clamped to 100)
Show-Http -Method Get -Uri "$baseUrl/notes?limit=10"

# 3. Get one note -> 200
Show-Http -Method Get -Uri "$baseUrl/notes/1"

# 4. Create a note -> 201 Created + Location header. We keep the Location for the next steps.
$body = @{ title = "My first note"; content = "Created from PowerShell" } | ConvertTo-Json
$r = Invoke-WebRequest -UseBasicParsing -Method Post -Uri "$baseUrl/notes" -ContentType "application/json" -Body $body
Write-Host ("Status: {0}" -f [int]$r.StatusCode)
Write-Host ("Location: {0}" -f $r.Headers["Location"])
Write-Host $r.Content
$noteUrl = $r.Headers["Location"]

# 5. Replace the whole note with PUT (title AND content) -> 200
$body = @{ title = "Replaced title"; content = "Replaced content" } | ConvertTo-Json
Show-Http -Method Put -Uri $noteUrl -Body $body

# 6. PUT with only the title -> 422 (PUT needs the complete resource)
$body = @{ title = "Only the title" } | ConvertTo-Json
Show-Http -Method Put -Uri $noteUrl -Body $body

# 7. Partial update with PATCH (only content) -> 200, the title stays the same
$body = @{ content = "Only the content changed" } | ConvertTo-Json
Show-Http -Method Patch -Uri $noteUrl -Body $body

# 8. Delete the note -> 204 No Content (empty body)
Show-Http -Method Delete -Uri $noteUrl

# 9. Get the deleted note -> 404 (application/problem+json)
Show-Http -Method Get -Uri $noteUrl

# --- Errors (all of them are application/problem+json) ---

# 10. Wrong Content-Type -> 415
Show-Http -Method Post -Uri "$baseUrl/notes" -Body "title=bad" -ContentType "text/plain"

# 11. Malformed JSON -> 400
Show-Http -Method Post -Uri "$baseUrl/notes" -Body '{"title": '

# 12. Valid JSON, invalid data (empty title, content is a number) -> 422 with an "errors" list
$body = @{ title = ""; content = 42 } | ConvertTo-Json
Show-Http -Method Post -Uri "$baseUrl/notes" -Body $body

# 13. Valid JSON that is not an object -> 422
Show-Http -Method Post -Uri "$baseUrl/notes" -Body '[1,2]'

# 14. Method not allowed -> 405 (look at the Allow header)
Show-Http -Method Delete -Uri "$baseUrl/notes"

# 15. Invalid limit -> 400
Show-Http -Method Get -Uri "$baseUrl/notes?limit=0"
