# Exercise 12: Roles & Permissions (PowerShell)

# 0. Start the app with a known admin password (or use the one printed at start-up):
#    $env:ADMIN_PASSWORD="admin123"
#    fastapi dev app.py

$baseUrl = "http://127.0.0.1:8000"
$headers = @{
    "Content-Type" = "application/json"
}

# 1. Register a student (new users are always students)
$studentBody = @{
    username = "student1"
    password = "pass1234"
} | ConvertTo-Json

Invoke-RestMethod -Method Post -Uri "$baseUrl/register" -Headers $headers -Body $studentBody

# 2. Try to register as admin: the role in the body is ignored, you get "role": "student"
$malloryBody = @{
    username = "mallory"
    password = "pass1234"
    role = "admin"
} | ConvertTo-Json

Invoke-RestMethod -Method Post -Uri "$baseUrl/register" -Headers $headers -Body $malloryBody

# 3. Login as admin (/login expects a JSON body)
$adminBody = @{
    username = "admin"
    password = "admin123"
} | ConvertTo-Json

$adminToken = (Invoke-RestMethod -Method Post -Uri "$baseUrl/login" -Headers $headers -Body $adminBody).access_token

# 4. Login as student
$studentToken = (Invoke-RestMethod -Method Post -Uri "$baseUrl/login" -Headers $headers -Body $studentBody).access_token

$adminAuthHeaders = @{
    Authorization = "Bearer $adminToken"
    "Content-Type" = "application/json"
}
$studentAuthHeaders = @{
    Authorization = "Bearer $studentToken"
    "Content-Type" = "application/json"
}

# 5. Admin dashboard as admin (expected: 200)
Invoke-RestMethod -Method Get -Uri "$baseUrl/admin/dashboard" -Headers $adminAuthHeaders

# 6. Admin dashboard as student (expected: 403)
try {
    Invoke-RestMethod -Method Get -Uri "$baseUrl/admin/dashboard" -Headers $studentAuthHeaders
} catch {
    Write-Host "Expected error: $($_.Exception.Message)"
}

# 7. Student data as student (expected: 200)
Invoke-RestMethod -Method Get -Uri "$baseUrl/student/data" -Headers $studentAuthHeaders

# 8. List users with pagination
Invoke-RestMethod -Method Get -Uri "$baseUrl/users?page=1&per_page=5" -Headers $adminAuthHeaders

# 9. Try to update a user as a student (expected: 403)
$hackBody = @{ password = "hacked-pass-1" } | ConvertTo-Json
try {
    Invoke-RestMethod -Method Put -Uri "$baseUrl/users/admin" -Headers $studentAuthHeaders -Body $hackBody
} catch {
    Write-Host "Expected error: $($_.Exception.Message)"
}

# 10. Promote student1 to admin (admins only) - from now on student1's token gets admin access
$promoteBody = @{ role = "admin" } | ConvertTo-Json
Invoke-RestMethod -Method Put -Uri "$baseUrl/users/student1" -Headers $adminAuthHeaders -Body $promoteBody

# 11. Delete a user (admins only)
Invoke-RestMethod -Method Delete -Uri "$baseUrl/users/mallory" -Headers $adminAuthHeaders
