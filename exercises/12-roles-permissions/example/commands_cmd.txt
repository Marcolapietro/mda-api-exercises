REM Exercise 12: Roles & Permissions (Windows CMD)

REM 0. Start the app with a known admin password (or copy the one printed at start-up):
REM    set ADMIN_PASSWORD=admin123
REM    fastapi dev app.py

REM 1. Register a student (new users are always students)
curl -X POST -H "Content-Type: application/json" -d "{\"username\":\"student1\",\"password\":\"pass1234\"}" http://127.0.0.1:8000/register

REM 2. Try to register as admin: the role in the body is ignored, you get "role": "student"
curl -X POST -H "Content-Type: application/json" -d "{\"username\":\"mallory\",\"password\":\"pass1234\",\"role\":\"admin\"}" http://127.0.0.1:8000/register

REM 3. Login as admin (copy the access_token as TOKEN_ADMIN)
curl -X POST -H "Content-Type: application/json" -d "{\"username\":\"admin\",\"password\":\"admin123\"}" http://127.0.0.1:8000/login

REM 4. Login as student (copy the access_token as TOKEN_STUDENT)
curl -X POST -H "Content-Type: application/json" -d "{\"username\":\"student1\",\"password\":\"pass1234\"}" http://127.0.0.1:8000/login

set TOKEN_ADMIN=PASTE_ADMIN_TOKEN_HERE
set TOKEN_STUDENT=PASTE_STUDENT_TOKEN_HERE

REM 5. Admin dashboard as admin (expected: 200)
curl -X GET -H "Authorization: Bearer %TOKEN_ADMIN%" http://127.0.0.1:8000/admin/dashboard

REM 6. Admin dashboard as student (expected: 403)
curl -i -X GET -H "Authorization: Bearer %TOKEN_STUDENT%" http://127.0.0.1:8000/admin/dashboard

REM 7. Student data as student (expected: 200)
curl -X GET -H "Authorization: Bearer %TOKEN_STUDENT%" http://127.0.0.1:8000/student/data

REM 8. List users with pagination
curl -X GET -H "Authorization: Bearer %TOKEN_ADMIN%" "http://127.0.0.1:8000/users?page=1&per_page=5"

REM 9. Try to update a user as a student (expected: 403)
curl -i -X PUT -H "Content-Type: application/json" -H "Authorization: Bearer %TOKEN_STUDENT%" -d "{\"password\":\"hacked-pass-1\"}" http://127.0.0.1:8000/users/admin

REM 10. Promote student1 to admin (admins only) - from now on student1's token gets admin access
curl -X PUT -H "Content-Type: application/json" -H "Authorization: Bearer %TOKEN_ADMIN%" -d "{\"role\":\"admin\"}" http://127.0.0.1:8000/users/student1

REM 11. Delete a user (admins only)
curl -X DELETE -H "Authorization: Bearer %TOKEN_ADMIN%" http://127.0.0.1:8000/users/mallory

REM 12. Try to delete your own admin account (expected: 409)
curl -i -X DELETE -H "Authorization: Bearer %TOKEN_ADMIN%" http://127.0.0.1:8000/users/admin
