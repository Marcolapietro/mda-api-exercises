# Exercise 9: API Key Authentication Testing Commands (PowerShell)

$baseUrl = "http://127.0.0.1:8000"
$headers = @{
    "Content-Type" = "application/json"
}

# Helper: show the HTTP status code and body of a failed request
function Show-HttpError($err, $label) {
    $status = [int]$err.Exception.Response.StatusCode
    Write-Host "$label -> HTTP $status $($err.ErrorDetails.Message)" -ForegroundColor Magenta
}

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "Exercise 9: API Key Authentication Tests" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

# ============================================================================
# PUBLIC ENDPOINTS (No Authentication Required)
# ============================================================================

Write-Host "==> 1. Registering user 'alice'..." -ForegroundColor Yellow
$registerBody = @{
    username = "alice"
    password = "secret123"
} | ConvertTo-Json

try {
    $registerResponse = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/register" -Headers $headers -Body $registerBody
    Write-Host "Registration successful!" -ForegroundColor Green
    Write-Host "Username: $($registerResponse.username)" -ForegroundColor White
    Write-Host "API Key: $($registerResponse.api_key)" -ForegroundColor White
    Write-Host "Message: $($registerResponse.message)" -ForegroundColor White
    Write-Host ""

    # Save API key for subsequent tests (the server will never show it again)
    $apiKey = $registerResponse.api_key

} catch {
    Show-HttpError $_ "Registration failed (user may already exist)"
    Write-Host "The server cannot show a key again (it only stores the hash). Restart the server (users live in memory) and run this script again." -ForegroundColor Yellow
    exit
}

# ============================================================================
# API KEY PROTECTED ENDPOINTS
# ============================================================================

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "API KEY PROTECTED ENDPOINTS" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

Write-Host "==> 2. Getting users list with API key (header X-API-Key)..." -ForegroundColor Yellow
$apiKeyHeaders = @{
    "X-API-Key" = $apiKey
}

try {
    $usersResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers $apiKeyHeaders
    Write-Host "Users list retrieved successfully:" -ForegroundColor Green
    $usersResponse | ConvertTo-Json | Write-Host
    Write-Host ""
} catch {
    Show-HttpError $_ "Error getting users"
    Write-Host ""
}

# ============================================================================
# KEY ROTATION AND REVOCATION (also protected by the API key)
# ============================================================================

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "ROTATE / REVOKE (protected by the API key)" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

Write-Host "==> 3. Rotating the API key (the current key identifies the user)..." -ForegroundColor Yellow
$oldApiKey = $apiKey

try {
    $rotateResponse = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/api-key/rotate" -Headers $apiKeyHeaders
    Write-Host "API key rotated:" -ForegroundColor Green
    $rotateResponse | ConvertTo-Json | Write-Host
    $apiKey = $rotateResponse.api_key
    Write-Host ""
} catch {
    Show-HttpError $_ "Error rotating API key"
    Write-Host ""
}

Write-Host "==> 4. Using the OLD key (should fail with 401)..." -ForegroundColor Yellow
try {
    Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers @{ "X-API-Key" = $oldApiKey } | Out-Null
    Write-Host "Unexpected success! The old key should no longer work." -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - old key rejected"
}
Write-Host ""

Write-Host "==> 5. Using the NEW key (should work)..." -ForegroundColor Yellow
try {
    $newKeyResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers @{ "X-API-Key" = $apiKey }
    Write-Host "New key works! Users: $($newKeyResponse.users -join ', ')" -ForegroundColor Green
} catch {
    Show-HttpError $_ "Error with new key"
}
Write-Host ""

# ============================================================================
# ERROR TESTING
# ============================================================================

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "ERROR TESTING" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

Write-Host "==> 6. Testing with missing API key (should fail)..." -ForegroundColor Yellow
try {
    $missingKeyResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users"
    Write-Host "Unexpected success! Should have failed." -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - API key missing"
    Write-Host ""
}

Write-Host "==> 7. Testing with invalid API key (should fail)..." -ForegroundColor Yellow
$invalidHeaders = @{
    "X-API-Key" = "sk_invalid-key-12345"
}

try {
    $invalidKeyResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers $invalidHeaders
    Write-Host "Unexpected success! Should have failed." -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - Invalid API key"
    Write-Host ""
}

Write-Host "==> 8. Rotating without a key (should fail)..." -ForegroundColor Yellow
try {
    Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/api-key/rotate" | Out-Null
    Write-Host "Unexpected success! Should have failed." -ForegroundColor Red
} catch {
    Show-HttpError $_ "Expected error - no key"
    Write-Host ""
}

# ============================================================================
# REGISTER SECOND USER
# ============================================================================

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "MULTIPLE USERS TEST" -ForegroundColor Cyan
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

Write-Host "==> 9. Registering second user 'bob'..." -ForegroundColor Yellow
$registerBody2 = @{
    username = "bob"
    password = "password456"
} | ConvertTo-Json

try {
    $registerResponse2 = Invoke-RestMethod -UseBasicParsing -Method Post -Uri "$baseUrl/register" -Headers $headers -Body $registerBody2
    Write-Host "Second user registered successfully!" -ForegroundColor Green
    Write-Host "Username: $($registerResponse2.username)" -ForegroundColor White
    Write-Host "API Key: $($registerResponse2.api_key)" -ForegroundColor White
    Write-Host ""

    $bobApiKey = $registerResponse2.api_key

    # Test bob's API key
    Write-Host "==> 10. Testing bob's API key..." -ForegroundColor Yellow
    $bobHeaders = @{
        "X-API-Key" = $bobApiKey
    }

    $bobUsersResponse = Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers $bobHeaders
    Write-Host "Bob's API key works! Users visible:" -ForegroundColor Green
    $bobUsersResponse | ConvertTo-Json | Write-Host
    Write-Host ""

    Write-Host "==> 11. Revoking bob's key (DELETE /api-key)..." -ForegroundColor Yellow
    $revokeResponse = Invoke-RestMethod -UseBasicParsing -Method Delete -Uri "$baseUrl/api-key" -Headers $bobHeaders
    Write-Host "$($revokeResponse.message)" -ForegroundColor Green
    try {
        Invoke-RestMethod -UseBasicParsing -Method Get -Uri "$baseUrl/users" -Headers $bobHeaders | Out-Null
        Write-Host "Unexpected success! The revoked key should not work." -ForegroundColor Red
    } catch {
        Show-HttpError $_ "Expected error - revoked key rejected"
    }
    Write-Host ""

} catch {
    Show-HttpError $_ "Error with second user"
    Write-Host ""
}

# ============================================================================
# SUMMARY
# ============================================================================

Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host "All tests completed!" -ForegroundColor Green
Write-Host ("=" * 70) -ForegroundColor Cyan
Write-Host ""

Write-Host "Summary:" -ForegroundColor Cyan
Write-Host "- API keys generated with secrets.token_urlsafe(32), prefix sk_" -ForegroundColor White
Write-Host "- Server stores only SHA-256(key) and shows the key ONCE" -ForegroundColor White
Write-Host "- /users endpoint: Protected by API key ONLY (X-API-Key header)" -ForegroundColor White
Write-Host "- POST /api-key/rotate and DELETE /api-key: Protected by the API key too (no body)" -ForegroundColor White
Write-Host "- Leaked key? Rotate it; the old one stops working" -ForegroundColor White
Write-Host ""

Write-Host "Your current API keys (store them in env vars, never in code):" -ForegroundColor Cyan
Write-Host "- alice: $apiKey" -ForegroundColor White
Write-Host "- bob: revoked" -ForegroundColor White
Write-Host ""
