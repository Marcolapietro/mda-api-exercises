REM Exercise 9: Pagination - Notes API (Windows CMD)
REM Start the API first: python app.py   (it starts with 95 seed notes: ids 1..95, newest first)
REM Always put URLs that contain & between double quotes in CMD.
REM -i shows the status line and the headers (look at the Link header).

REM --- Offset pagination: ?offset=N&limit=M ---

REM 1. First page of 5 -> 200, notes 95..91, page.total = 95, Link: rel="next"
curl -i "http://127.0.0.1:5000/notes?offset=0&limit=5"

REM 2. A page in the middle (the slide's example) -> notes 55..36, Link with rel="next" and rel="prev"
curl -i "http://127.0.0.1:5000/notes?offset=40&limit=20"

REM 3. Last page -> notes 5..1, page.next = null, only rel="prev"
curl -i "http://127.0.0.1:5000/notes?offset=90&limit=5"

REM --- Cursor pagination: ?cursor=<opaque>&limit=M ---

REM 4. First page (no cursor) -> 200, notes 95..91, page.next_cursor = "OTE" (base64 of "91")
curl -i "http://127.0.0.1:5000/notes?limit=5"

REM 5. Next page using the cursor -> notes 90..86
curl -i "http://127.0.0.1:5000/notes?cursor=OTE&limit=5"

REM 6. Default page size (20) and maximum (100: limit=500 is clamped to 100)
curl -i "http://127.0.0.1:5000/notes"
curl -i "http://127.0.0.1:5000/notes?limit=500"

REM --- Errors (application/problem+json) ---

REM 7. Mixing offset and cursor -> 400
curl -i "http://127.0.0.1:5000/notes?offset=0&cursor=OTE"

REM 8. Invalid cursor -> 400
curl -i "http://127.0.0.1:5000/notes?cursor=not-a-real-cursor"

REM 9. Invalid limit / offset -> 400
curl -i "http://127.0.0.1:5000/notes?limit=0"
curl -i "http://127.0.0.1:5000/notes?offset=-1"

REM --- Experiment: insert a note while you paginate ---

REM 10. OFFSET. Read page 1, insert a note, read page 2.
REM     The note that was last on page 1 shows up AGAIN as the first one on page 2.
curl "http://127.0.0.1:5000/notes?offset=0&limit=5"
curl -X POST -H "Content-Type: application/json" -d "{\"title\":\"Inserted while paginating\"}" http://127.0.0.1:5000/notes
curl "http://127.0.0.1:5000/notes?offset=5&limit=5"

REM 11. CURSOR. Read page 1 and copy its "next_cursor".
curl "http://127.0.0.1:5000/notes?limit=5"

REM     Replace PASTE_NEXT_CURSOR_HERE with the next_cursor you just got, then insert a note and read page 2.
REM     Page 2 continues right after the last note of page 1: no duplicates.
set CURSOR=PASTE_NEXT_CURSOR_HERE
curl -X POST -H "Content-Type: application/json" -d "{\"title\":\"Inserted while paginating (cursor)\"}" http://127.0.0.1:5000/notes
curl "http://127.0.0.1:5000/notes?cursor=%CURSOR%&limit=5"
